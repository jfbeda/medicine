\section{Abdominal Examination}
\label{sec:abdominal-examination}


\begin{itemize}
    \item Introduction (Name, date of birth, chaperone, any pain)
    \item End of the bed
    \begin{itemize}
        \item Bedside equipment, jaundice, pallor, cachexia, distension
    \end{itemize}
    \item Hands and arms
    \begin{itemize}
        \item Hands (erythema, Dupuytren's contracture, asterixis, nicotine)
        \item Nails (clubbing, koilonychia, leukonychia)
        \item Radial pulse
    \end{itemize}
    \item Face, mouth, neck, and upper chest
    \begin{itemize}
        \item Eyes (jaundice in sclera, pallor in eyelids)
        \item Mouth (angular stomatitis, glossitis, candidiasis, dentation, dehydration)
        \item Lymph nodes (cervical, supraclavicular, Virkow's node)
    \end{itemize}
    \item Chest and Abdomen
    \begin{itemize}
        \item Spider nevae, striae, scars
        \item Cough (hernia)
    \end{itemize}
    \item Palpation
    \begin{itemize}
        \item Confirm any pain
        \item Nine regions (light then firm palpation)
        \item Liver (feel liver edge by moving the hand towards costal margin)
        \item Spleen (feel diagonally towards left costal margin)
        \item Kidneys (bounce them back and forth)
        \item Abdominal aorta 
    \end{itemize}
    \item Percussion
    \begin{itemize}
        \item Nine regions
        \item Liver (upper and lower borders)
        \item Ascites (shifting dullness)
    \end{itemize}
    \item Auscultation
    \begin{itemize}
        \item Bowel sounds
        \item Aortic bruits, renal bruits, iliac/femoral bruits
    \end{itemize}
\end{itemize}

\subsection{Respiratory Examination}

\begin{itemize}
    \item Introduction (Name, date of birth, chaperone, any pain)
    \item End of the bed
    \begin{itemize}
        \item Bedside equipment, jaundice, pallor, cyanosis, 
    \end{itemize}
    \item Hands and arms
    \begin{itemize}
        \item Hands (cyanosis, nicotine, asterixis, tremor, temperature)
        \item Nails (clubbing, koilonychia)
        \item Radial pulse, respiratory rate, blood pressure
    \end{itemize}
    \item Face, mouth, neck, and upper chest
    \begin{itemize}
        \item Eyes (pallor, Horner’s syndrome)
        \item Jugular venous pressure (JVP)
        \item Mouth (cyanosis, candidiasis)
        \item Trachial deviation
    \end{itemize}    
    \item Lymph nodes
    \begin{itemize}
        \item Submental, submandibular, pre/post-auricular, superficial/deep cervical, posterior cervical, supraclavicular, oxipital
    \end{itemize}
    \item Chest and Abdomen
    \begin{itemize}
        \item Scars, asymmetry, chest shape, use of accessory muscles, respiratory rate, barrel chest
        \item Chest expansion (front \& back)
        \item Chest percussion (front \& back)
        \item Chest auscultation (front \& back)
        \item Chest 99 (front \& back)
    \end{itemize}
    \item Oedema
    \begin{itemize}
        \item Sacral, pedal
    \end{itemize}
\end{itemize}
