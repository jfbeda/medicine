\section{Registering with Learning@Wales}

\begin{enumerate}
    \item Go to
    \href{https://learning.nhs.wales/}{\texttt{learning.nhs.wales}}.

    \item Select \term{Learning@Wales}.

    \item Select the button right below the text that says

    \begin{quote}
        \enquote{Please click the button below to register for an
        account if you are unable to self-register.}
    \end{quote}
\end{enumerate}
